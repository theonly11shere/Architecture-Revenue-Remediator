import os
import smtplib
from email.mime.text import MIMEText
from fastapi import APIRouter, HTTPException

# Use your existing router instance or app instance
router = APIRouter()


@router.get("/test-email")
def test_email(to: str):
  smtp_host = os.getenv("SMTP_HOST")
  smtp_port = int(os.getenv("SMTP_PORT", 587))
  smtp_user = os.getenv("SMTP_USER") or os.getenv("SMTP_USERNAME")
  smtp_password = os.getenv("SMTP_PASSWORD")
  email_from = os.getenv("EMAIL_FROM") or smtp_user

  if not smtp_host or not smtp_user or not smtp_password:
    raise HTTPException(
        status_code=500, detail="Missing SMTP environment variables in Railway."
    )

  try:
    msg = MIMEText(
        "This is a test message to confirm your audit scanner email integration"
        " is working."
    )
    msg["Subject"] = "Audit Scanner Test Email"
    msg["From"] = email_from
    msg["To"] = to

    with smtplib.SMTP(smtp_host, smtp_port) as server:
      server.starttls()
      server.login(smtp_user, smtp_password)
      server.sendmail(email_from, [to], msg.as_string())

    return {"status": "success", "message": f"Test email successfully sent to {to}"}
  except Exception as e:
    raise HTTPException(
        status_code=500, detail=f"SMTP Error: {str(e)}"
    )
